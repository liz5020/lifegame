// ========== 三、3.8 幸福感系統（9/20新增，取代原七、7.1.4裡的幸福感角色定義）==========
// 幸福感是系統裡唯一量「過得好不好」而非「做到了什麼」的軸，不產生任何判決。
// 三層：基準線（長期水準，由五條來源管道決定）＋事件即時波動（AI只給emotional_tone，程式查表）＋朝基準線回歸。
// 以下數字全部是【測試參數,待驗證】，由模擬校準，不得由代碼自行升格為定案（3.8.3、3.8.4）
const HAPPINESS_TONE_DELTA = { uplifting:[6,12], warm:[2,5], unsettling:[-5,-2], heavy:[-12,-6] };
const HAPPINESS_TONE_ENUM = Object.keys(HAPPINESS_TONE_DELTA);
const HAPPINESS_REGRESSION_INTERVAL = 5;        // 每N回合朝基準線回歸一次
const HAPPINESS_REGRESSION_RATIO = 0.25;        // 每次移動的比例
const HAPPINESS_CHANNEL_TOP_WEIGHTS = [1.0, 0.7, 0.4]; // 3.8.2【定案】取前三高加權，第四第五條不計入
const HAPPINESS_CHANNEL_DIMINISH_SCALE = 55;    // 管道邊際遞減的尺度參數
const HAPPINESS_LOW_THRESHOLD = 35;             // 4.5 longest_low_streak 的低谷閾值（設計文件標【待補充】，這裡先給第一輪測試值）
const AUTONOMY_NEUTRAL_DEFAULT = 50;            // 還沒遇到任何可跳過里程碑時的中性值
// 3.8.4「可跳過的里程碑」在v1的實作範圍：只收「玩家自己這一生要不要走這條路」的選擇。
// parent_retirement（父母退休）不是玩家的選擇、graduation_*等auto里程碑目前代碼裡不存在skipped路徑，兩者皆不列入。
// ⚠️落差：first_child(生育/領養)、home_purchase(買房)、moved_out(搬出家裡)在語意上是可跳過的傳統路線，
// 但現行代碼只有「達成時自動completed」沒有「明確決定不做」的skipped入口，暫時無法計入自主感，已記入QA手冊34.2
const AUTONOMY_MILESTONES = ["marriage_decision","first_startup","first_credit_card","first_part_time_job"];
const INTIMATE_RELATION_RE = /配偶|伴侶|先生|太太|老公|老婆|兒子|女兒|父親|母親|爸|媽|哥|姊|姐|弟|妹/;

function happinessChannelDiminish(raw){
  // 3.8.2【定案】每條管道各自套用3.5.3的邊際遞減。3.5.3的成長式 Δ=原始Δ×(100−x)/100 連續化後
  // 封閉解為 x(t)=100(1−e^(−t/scale))，這裡直接拿它當「原始投入→管道分數」的靜態轉換：
  // 高位投入換來的分數越來越少，且永遠逼近但不撞滿100。
  if(!(raw>0)) return 0;
  return clamp(100*(1-Math.exp(-raw/HAPPINESS_CHANNEL_DIMINISH_SCALE)), 0, 100);
}
function avgAffinity(list){
  if(!list || !list.length) return 0;
  return list.reduce((a,c)=>a+(c.affinity||0),0)/list.length;
}
function computeAutonomyRaw(s){
  // 3.8.4【定案】chosen佔「已遇到的可跳過里程碑」的比例。
  // not_offered（未觸發）與unavailable（條件未達成）都不進分母——設計文件明講「一直沒遇到對象的角色則不加分」，
  // 不加分不等於扣分，若讓它們稀釋比例就變成扣分，會直接違反六、6.4中立性。
  const m = s.milestones||{};
  let chosen=0, encountered=0;
  AUTONOMY_MILESTONES.forEach(id=>{
    const st = m[id];
    if(st==="skipped"){ chosen++; encountered++; }
    else if(st==="completed" || st==="active"){ encountered++; }
  });
  if(!encountered) return AUTONOMY_NEUTRAL_DEFAULT;
  return (chosen/encountered)*100;
}
function milestoneSkipReasonFor(s, id){
  // 3.8.4【定案】skip_reason三選一。completed/active代表這條路實際走了，不屬於「跳過」，回null不記錄。
  const st = (s.milestones||{})[id];
  if(st==="skipped") return "chosen";
  if(st==="locked") return "unavailable";
  if(st==="available") return "not_offered";
  return null;
}
function refreshMilestoneSkipReasons(s){
  s.milestoneSkipReason = s.milestoneSkipReason || {};
  MILESTONE_DEFS.forEach(def=>{
    const reason = milestoneSkipReasonFor(s, def.id);
    if(reason) s.milestoneSkipReason[def.id] = reason;
    else delete s.milestoneSkipReason[def.id];
  });
}
function computeHappinessChannels(s){
  // 3.8.2【定案】五條來源管道，各自0-100、各自套用邊際遞減。
  // ⚠️落差：「社團參與」與「職涯認同」目前沒有對應的結構化欄位，v1先由人脈／才識承擔，已記入QA手冊34.2
  const chars = s.characters||[];
  const intimate = chars.filter(c=>INTIMATE_RELATION_RE.test(c.relation||""));
  const others = chars.filter(c=>!INTIMATE_RELATION_RE.test(c.relation||""));
  const interests = s.interestCandidates||[];
  const topInterest = interests.length ? Math.max.apply(null, interests.map(c=>c.investment||0)) : 0;
  const finHealth = computeFinancialHealthForState(s);
  const raw = {
    intimacy: avgAffinity(intimate),
    belonging: (s.stats.network||0)*0.6 + avgAffinity(others)*0.4,
    selfActualization: (s.stats.knowledge||0)*0.5 + clamp(topInterest,0,100)*0.5,
    stability: (s.stats.health||0)*0.5 + finHealth*0.5,
    autonomy: computeAutonomyRaw(s)
  };
  const out = {};
  Object.keys(raw).forEach(k=>{ out[k] = happinessChannelDiminish(raw[k]); });
  return out;
}
function computeHappinessBaseline(s){
  // 3.8.2【定案】取前三高加權，正規化0-100。第四、第五條不計入——
  // 這是六、6.4中立性的機制保證：管道數量不足本身不扣分，三條撐得夠高就跟走傳統路線的人一樣能到滿值。
  const ch = computeHappinessChannels(s);
  const sorted = Object.keys(ch).map(k=>ch[k]).sort((a,b)=>b-a).slice(0,3);
  let num=0, den=0;
  sorted.forEach((v,i)=>{ num += v*HAPPINESS_CHANNEL_TOP_WEIGHTS[i]; den += HAPPINESS_CHANNEL_TOP_WEIGHTS[i]; });
  return den ? clamp(num/den, 0, 100) : 50;
}
function happinessToneDelta(tone){
  // 3.8.3【定案】AI回傳enum以外的值，一律視為中性（增減量0），不崩潰也不憑猜測賦值
  const range = HAPPINESS_TONE_DELTA[tone];
  if(!range) return 0;
  return rnd(range[0], range[1]);
}
function applyHappiness(s, tone){
  // 3.8.4【定案,程式層禁令】這裡的輸入只有「事件結果(tone)」與「來源管道的投入回饋(baseline)」兩種，
  // 不得以「某里程碑是否達成／某狀態是否成立」作為加減條件。QA可直接靜態檢查本函式。
  s.happiness = clamp(s.happiness + happinessToneDelta(tone), 0, 100);
  s.happinessBaseline = computeHappinessBaseline(s);
  s.happinessRegressionCounter = (s.happinessRegressionCounter||0) + 1;
  if(s.happinessRegressionCounter >= HAPPINESS_REGRESSION_INTERVAL){
    s.happinessRegressionCounter = 0;
    s.happiness = clamp(s.happiness + (s.happinessBaseline - s.happiness)*HAPPINESS_REGRESSION_RATIO, 0, 100);
  }
}

// ========== 四、4.5 人生軌跡紀錄 life_trajectory（9/20新增）==========
// 【定案】每個人生階段結束時由客戶端結算一次並凍結，不可回頭修改，AI不參與結算。
// 階段切法沿用4.2.3既有定義（chronicleLifeStageKey：學生時期＋2.5七個年齡帶，共8段）
const TRAJECTORY_KEY_NPC_COUNT = 3; // 4.5【定案】關係值最高的2至3位，取上限3
function spectrumSnapshot(s){
  // 依附兩軸（5.2.3/12.1.4）＋同儕位置（5.2.4）＋做事態度五條（3.7），共8個值
  const c = s.conscientiousness||{};
  return {
    anxiety: Math.round(s.anxiety||0), avoidance: Math.round(s.avoidance||0),
    peerPosition: Math.round(s.peerPosition||0),
    selfDiscipline: Math.round(c.selfDiscipline||0), prudence: Math.round(c.prudence||0),
    achievement: Math.round(c.achievement||0), responsibility: Math.round(c.responsibility||0),
    teamSolo: Math.round(c.teamSolo||0)
  };
}
function newStageAccum(s){
  return { stage: chronicleLifeStageKey(s), ageFrom: s.age, turns:0,
    happinessSum:0, peak:s.happiness, low:s.happiness,
    lowStreak:0, longestLowStreak:0, channelSums:{},
    chronicleStart:(s.chronicle||[]).length, breakthroughEvents:[] };
}
function freezeStageTrajectory(s){
  const a = s.stageAccum;
  if(!a || !a.turns) return;
  const chars = (s.characters||[]).slice().sort((x,y)=>(y.affinity||0)-(x.affinity||0));
  let dominant=null, best=-1;
  Object.keys(a.channelSums).forEach(k=>{ if(a.channelSums[k]>best){ best=a.channelSums[k]; dominant=k; } });
  s.lifeTrajectory.push({
    stage: a.stage, age_from: a.ageFrom, age_to: s.age, turns: a.turns,
    happiness_avg: +(a.happinessSum/a.turns).toFixed(1),
    happiness_peak: Math.round(a.peak), happiness_low: Math.round(a.low),
    dominant_channel: dominant,
    longest_low_streak: a.longestLowStreak,
    spectrum_snapshot: spectrumSnapshot(s),
    resume_entries: (s.chronicle||[]).slice(a.chronicleStart),
    key_npcs: chars.slice(0,TRAJECTORY_KEY_NPC_COUNT).map(c=>({ name:c.name, relation:c.relation||"", affinity:Math.round(c.affinity||0) })),
    // ⚠️落差：六、6.2破格里程碑目前只存在於system prompt的敘事指示，沒有結構化觸發訊號可供客戶端記錄，
    // 這個欄位先保留空陣列，等6.2實作出可判定的光譜位移觸發點再填（已記入QA手冊34.2）
    breakthrough_events: a.breakthroughEvents.slice()
  });
}
function updateLifeTrajectory(s, forceEnding){
  s.lifeTrajectory = s.lifeTrajectory || [];
  if(!s.stageAccum) s.stageAccum = newStageAccum(s);
  const nowStage = chronicleLifeStageKey(s);
  if(nowStage !== s.stageAccum.stage){ freezeStageTrajectory(s); s.stageAccum = newStageAccum(s); }
  const a = s.stageAccum;
  a.turns++;
  a.happinessSum += s.happiness;
  if(s.happiness > a.peak) a.peak = s.happiness;
  if(s.happiness < a.low) a.low = s.happiness;
  if(s.happiness < HAPPINESS_LOW_THRESHOLD){
    a.lowStreak++;
    if(a.lowStreak > a.longestLowStreak) a.longestLowStreak = a.lowStreak;
  } else a.lowStreak = 0;
  const ch = computeHappinessChannels(s);
  Object.keys(ch).forEach(k=>{ a.channelSums[k] = (a.channelSums[k]||0) + ch[k]; });
  if(forceEnding){ freezeStageTrajectory(s); s.stageAccum = null; }
}

// ========== 七、7.1.4 死亡結局的人生總結（9/20整節改版，取代ENDING_DEFINITIONS標籤計分制）==========
// 【定案】不輸出任何標籤、分數、排名。素材的挑選與排序全由客戶端從life_trajectory完成，AI只負責寫成回顧的語氣。
const SUMMARY_TONE_BANDS = [ // 7.1.4.2【定案】敘事溫度四級由happiness_avg決定；分界數字為【測試參數】
  { min:65, tone:"釋然" }, { min:45, tone:"平穩" }, { min:25, tone:"惋惜" }, { min:0, tone:"沉重" }
];
const SUMMARY_RESUME_PER_STAGE = 3;  // 7.1.4.2【待補充】每階段挑幾則履歷條目——第一輪測試值
const SUMMARY_NPC_PER_STAGE = 2;     // 7.1.4.2【待補充】每階段挑幾位角色卡——第一輪測試值
const SPECTRUM_SHIFT_THRESHOLD = 15; // 7.1.4.2【待補充】L2變化句的位移閾值——第一輪測試值
const SPECTRUM_LABELS = { anxiety:"焦慮軸", avoidance:"迴避軸", peerPosition:"同儕位置",
  selfDiscipline:"自律", prudence:"審慎", achievement:"成就傾向", responsibility:"責任感", teamSolo:"團隊vs單打獨鬥" };
const HAPPINESS_CHANNEL_LABELS = { intimacy:"親密與家庭", belonging:"友情與歸屬",
  selfActualization:"自我實現", stability:"身心安穩", autonomy:"自主感" };
function lifeStageLabel(key){
  if(key==="student") return "學生時期";
  if(key==="career_other") return "晚年";
  return key + "歲";
}
function summaryToneFor(avg){
  for(let i=0;i<SUMMARY_TONE_BANDS.length;i++){ if(avg>=SUMMARY_TONE_BANDS[i].min) return SUMMARY_TONE_BANDS[i].tone; }
  return SUMMARY_TONE_BANDS[SUMMARY_TONE_BANDS.length-1].tone;
}
function buildLifeSummaryMaterial(s){
  const traj = s.lifeTrajectory || [];
  const segments = traj.map(t=>({
    stage: t.stage, stage_label: lifeStageLabel(t.stage),
    age_from: t.age_from, age_to: t.age_to,
    tone: summaryToneFor(t.happiness_avg),
    dominant_channel: t.dominant_channel,
    dominant_channel_label: HAPPINESS_CHANNEL_LABELS[t.dominant_channel] || null,
    longest_low_streak: t.longest_low_streak,
    resume_entries: (t.resume_entries||[]).slice(0, SUMMARY_RESUME_PER_STAGE),
    key_npcs: (t.key_npcs||[]).slice(0, SUMMARY_NPC_PER_STAGE),
    breakthrough_events: t.breakthrough_events || []
  }));
  const transitions = [];
  for(let i=1;i<traj.length;i++){
    const a = traj[i-1].spectrum_snapshot || {}, b = traj[i].spectrum_snapshot || {};
    let axis=null, delta=0;
    Object.keys(SPECTRUM_LABELS).forEach(k=>{
      const d = (b[k]||0) - (a[k]||0);
      if(Math.abs(d) > Math.abs(delta)){ delta = d; axis = k; }
    });
    if(axis && Math.abs(delta) >= SPECTRUM_SHIFT_THRESHOLD){
      transitions.push({ after_stage: traj[i-1].stage,
        from_stage_label: lifeStageLabel(traj[i-1].stage), to_stage_label: lifeStageLabel(traj[i].stage),
        axis, axis_label: SPECTRUM_LABELS[axis], from: a[axis]||0, to: b[axis]||0,
        direction: delta>0 ? "up" : "down" });
    }
  }
  return { segments, transitions };
}
function assembleEnding(s, r){
  // 客戶端先挑好素材，再把AI寫的文字貼回對應的段落；AI沒寫到的段落留空字串，不補猜測內容
  const material = buildLifeSummaryMaterial(s);
  const ls = r.life_summary || {};
  const byStage = {}, byTrans = {};
  (ls.segments||[]).forEach(x=>{ if(x && x.stage) byStage[x.stage] = x.text || ""; });
  (ls.transitions||[]).forEach(x=>{ if(x && x.after_stage) byTrans[x.after_stage] = x.text || ""; });
  return {
    segments: material.segments.map(m=>({ stage:m.stage, label:m.stage_label, age_from:m.age_from, age_to:m.age_to,
      tone:m.tone, resume_entries:m.resume_entries, key_npcs:m.key_npcs, text: byStage[m.stage] || "" })),
    transitions: material.transitions.map(t=>({ after_stage:t.after_stage, text: byTrans[t.after_stage] || "" })),
    epitaph: ls.epitaph || "",
    successionAvailable: !!r.succession_available
  };
}

3. **3.8.4 程式層禁令的靜態檢查**（設計文件明寫 QA 可直接查代碼）：搜尋是否存在「以里程碑／狀態是否成立作為幸福感加減條件」的寫法，例如 `happiness` 與 `milestones` 出現在同一個條件式裡。預期結果：`applyHappiness()` 是唯一改動 `s.happiness` 的地方，輸入只有 tone 與 baseline。
4. **純函式驗證**（vm stub 跑實際 `index.html` 程式碼，不要重新實作）：
   - `happinessToneDelta()`：四個 enum 各自落在對應區間；`undefined`／`"有點warm"`／`null`／`123` 一律回 0，不 throw。
   - `computeHappinessBaseline()`：造三種角色對照——(a) 五條管道都中等、(b) 只有三條高（不成家：intimacy=0、其餘三條高）、(c) 五條都高。**驗證重點：(b) 的基準線必須跟 (c) 幾乎一樣**，這是六、6.4 中立性的機制保證，(b) 明顯低於 (c) 就是實作錯了。
   - `computeAutonomyRaw()`：marriage_decision=skipped 時自主感升高；全部 available（沒遇到）時回中性 50 而不是 0。
   - `applyHappiness()`：連跑 5 回合確認第 5 回合觸發回歸；幸福感遠低於基準線時回歸方向朝上、遠高於時朝下。

## (E) 要回報給使用者確認的判斷點

這幾條是實作時我自己下的判斷，不是設計文件裡的原文，請在回報裡列出來讓使用者否決：

1. **自主感的分母**：`not_offered` 與 `unavailable` 都不進分母。理由是設計文件明講「一直沒遇到對象的角色則不加分」，不加分不等於扣分；若讓它們稀釋比例就變成實質扣分，會直接違反六、6.4。
2. **`AUTONOMY_MILESTONES` 只收 4 個**：marriage_decision／first_startup／first_credit_card／first_part_time_job。`parent_retirement` 不是玩家的選擇所以排除。
3. **管道邊際遞減用的曲線**：3.5.3 的成長式連續化後的封閉解 `x=100(1−e^(−raw/55))`，scale=55 是測試參數。
4. **三個落差（已記進 QA 手冊 34.2，但值得再講一次）**：
   - `breakthrough_events` 永遠是空陣列。六、6.2 破格里程碑目前只存在於 system prompt 的敘事指示，沒有結構化觸發訊號可以讓客戶端記錄。要填這個欄位得先實作 6.2 的光譜位移觸發判定。
   - 3.8.2 的「社團參與」「職涯認同」沒有對應的結構化欄位，v1 先由人脈／才識承擔。
   - first_child（生育/領養）、home_purchase（買房）、moved_out（搬出家裡）語意上是可跳過的傳統路線，但現行代碼只有「達成時自動 completed」，沒有「明確決定不做」的 skipped 入口，所以算不進自主感。**這條最值得優先補**，因為六、6.4 最常被引用的例子就是「不成家」，而「不生小孩」目前在機制上是完全隱形的。
5. **大改動預告**：新增 `life_trajectory` 遊戲狀態結構、移除 `ENDING_DEFINITIONS`，**舊存檔會跑不動，需要清空重來**。
